library(tidymodels)
library(tidyverse)
library(glmnet) # Ridge, Lasso, Elastic Net engine
library(patchwork) # combining plots
library(vip) # variable importance plots
tidymodels_prefer()
set.seed(2025)
set.seed(2025)
n <- 2500
clv_data <- tibble(
  customer_id = paste0("CLV", str_pad(1:n, 5, pad = "0")),
  age = round(pmin(pmax(rnorm(n, mean = 42, sd = 12), 18), 85)),
  income = round(rlnorm(n, meanlog = 10.8, sdlog = 0.55)),
  tenure_months = round(pmax(1, rnorm(n, mean = 30, sd = 18))),
  num_products = sample(
    1:6,
    n,
    replace = TRUE,
    prob = c(0.25, 0.30, 0.20, 0.12, 0.08, 0.05)
  ),
  freq_purchases = rpois(n, lambda = 8),
  recency_days = round(pmax(0, rexp(n, rate = 1 / 30))), # mean = 1/rate
  avg_basket = round(rlnorm(n, meanlog = 4.0, sdlog = 0.45), 2), # $
  channel = sample(
    c("online", "in_store", "both"),
    n,
    replace = TRUE,
    prob = c(0.45, 0.25, 0.30)
  ),
  loyalty_tier = sample(
    c("Bronze", "Silver", "Gold", "Platinum"),
    n,
    replace = TRUE,
    prob = c(0.35, 0.30, 0.22, 0.13)
  ),
  region = sample(
    c("West", "South", "Midwest", "Northeast"),
    n,
    replace = TRUE
  ),
  promo_response = sample(
    c("yes", "no"),
    n,
    replace = TRUE,
    prob = c(0.40, 0.60)
  )
) |>
  mutate(
    # annual_spend — continuous outcome
    # driven by income, frequency, basket size, products owned, loyalty
    annual_spend = round(
      pmax(
        10,
        200 +
          0.004 * income +
          18 * freq_purchases +
          0.85 * avg_basket +
          45 * num_products -
          0.8 * recency_days +
          2.0 * tenure_months +
          case_when(
            loyalty_tier == "Platinum" ~ 350,
            loyalty_tier == "Gold" ~ 200,
            loyalty_tier == "Silver" ~ 80,
            TRUE ~ 0
          ) +
          if_else(channel == "both", 120, 0) +
          if_else(promo_response == "yes", 75, 0) +
          rnorm(n, 0, 180) # noise
      ),
      2
    ),
    # Introduce missingness
    income = if_else(runif(n) < 0.07, NA_real_, income),
    avg_basket = if_else(runif(n) < 0.05, NA_real_, avg_basket),
    # Encode factors
    channel = factor(channel),
    loyalty_tier = factor(
      loyalty_tier,
      levels = c("Bronze", "Silver", "Gold", "Platinum")
    ),
    region = factor(region),
    promo_response = factor(promo_response)
  )

Exercise 13.2: Elastic Net Mixing

clv_split <- initial_split(clv_data, prop = 0.75)
clv_train <- training(clv_split)
clv_test <- testing(clv_split)

clv_recipe <- recipe(annual_spend ~ age + income + tenure_months + num_products +
                        freq_purchases + recency_days + avg_basket + channel +
                        loyalty_tier + region + promo_response,
                      data = clv_train) |>
  step_impute_median(all_numeric_predictors()) |>
  step_dummy(all_nominal_predictors()) |>
  step_normalize(all_numeric_predictors())
mixture_values <- c(0.25, 0.50, 0.75)

fit_and_eval <- function(mix_val) {
  spec <- linear_reg(penalty = 0.1, mixture = mix_val) |>
    set_engine("glmnet")

  wf <- workflow() |>
    add_recipe(clv_recipe) |>
    add_model(spec)

  fit <- wf |> fit(data = clv_train)

  coefs <- tidy(fit)
  nonzero_count <- coefs |> filter(estimate != 0) |> nrow()

  preds <- predict(fit, clv_test) |> bind_cols(clv_test)
  test_rmse <- preds |> rmse(truth = annual_spend, estimate = .pred) |> pull(.estimate)

  tibble(mixture = mix_val, nonzero_coefs = nonzero_count, test_rmse = test_rmse)
}

en_results <- map_dfr(mixture_values, fit_and_eval)
en_results

Results: All three mixture values (0.25, 0.50, 0.75) produced identical results, 16 nonzero coefficients and a test RMSE of about 181. This suggests the penalty of 0.1 was too weak to meaningfully differentiate the mixture effect between Ridge-like and Lasso-like behavior.

Exercise 13.3: Polynomial Terms

poly_recipe_deg <- function(deg) {
  recipe(annual_spend ~ age + income + tenure_months + num_products +
           freq_purchases + recency_days + avg_basket + channel +
           loyalty_tier + region + promo_response,
         data = clv_train) |>
    step_impute_median(all_numeric_predictors()) |>
    step_poly(freq_purchases, tenure_months, degree = deg) |>
    step_dummy(all_nominal_predictors()) |>
    step_normalize(all_numeric_predictors())
}

lm_spec <- linear_reg() |> set_engine("lm")
folds <- vfold_cv(clv_train, v = 10)

get_cv_rmse <- function(deg) {
  wf <- workflow() |>
    add_recipe(poly_recipe_deg(deg)) |>
    add_model(lm_spec)

  res <- fit_resamples(wf, resamples = folds, metrics = metric_set(rmse))
  collect_metrics(res) |> filter(.metric == "rmse") |> mutate(degree = deg)
}

poly_results <- map_dfr(c(1, 2, 3), get_cv_rmse)
poly_results
deg3_spec <- linear_reg(penalty = 0.001, mixture = 0) |>
  set_engine("glmnet")

deg3_wf <- workflow() |>
  add_recipe(poly_recipe_deg(3)) |>
  add_model(deg3_spec)

deg3_fit <- deg3_wf |> fit(data = clv_train)

train_preds <- predict(deg3_fit, clv_train) |> bind_cols(clv_train)
test_preds <- predict(deg3_fit, clv_test) |> bind_cols(clv_test)

train_preds |> rmse(truth = annual_spend, estimate = .pred)
test_preds |> rmse(truth = annual_spend, estimate = .pred)

Results: CV RMSE stayed flat (around 188) across polynomial degrees 1, 2, and 3, well within one standard error of each other. Training RMSE (187) was close to test RMSE (183) for the degree-3 model, showing no overfitting. This suggests the true relationship between these predictors and spend is linear rather than curved.

Q1: Three Most Important Predictors

The three most important predictors of annual spend are income (134), loyalty tier (Platinum ~120, Gold ~77.7), and number of products (65.8), all positive. Income reflects disposable spending power, a fixed trait the business can’t change directly but can act on indirectly (income-sensitive discount thresholds, free shipping minimums). Loyalty tier likely reflects a two-way relationship: customers who already spend more reach higher tiers, and the perks tied to those tiers then incentivize continued spending to maintain status; unlike income, loyalty tier is something the business directly designs and controls. Number of products owned is positive because it broadens a customer’s engagement across categories, giving more reasons to return, actionable via cross-selling and bundling strategies.

Q2: A Predictor That Was Zeroed Out

The only predictor Lasso zeroed out in the final tuned model was age. Intuitively, age seems like it should matter, since younger and older customers likely have different purchasing power and spending habits. However, age’s effect is probably captured indirectly through income, since income tends to rise with career stage and age. Once income is already in the model, age doesn’t add much unique predictive information, so Lasso treated it as redundant rather than truly irrelevant, and excluded it in favor of keeping income.

Q3: Most Actionable Lever for a Marketing Manager

The single most actionable lever is promo response. Unlike traits such as income, age, or region, which are fixed characteristics the business can only observe, promo response reflects something the business directly controls, the promotions and offers it chooses to run. A marketing manager can launch a targeted campaign immediately and measure its effect on spend, making it the most practical and immediate lever available.
